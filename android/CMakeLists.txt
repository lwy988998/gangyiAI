cmake_minimum_required(VERSION 3.22)
project(GangyiAndroid LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT GANGYI_ANDROID_DEPS)
    message(FATAL_ERROR "请先运行 scripts/prepare_android_native.py 准备经过校验的安卓依赖")
endif()
file(TO_CMAKE_PATH "${GANGYI_ANDROID_DEPS}" GANGYI_ANDROID_DEPS)
set(FETCHCONTENT_SOURCE_DIR_CROW "${GANGYI_ANDROID_DEPS}/crow" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_NLOHMANN_JSON "${GANGYI_ANDROID_DEPS}/json" CACHE PATH "" FORCE)
set(ASIO_INCLUDE_DIR "${GANGYI_ANDROID_DEPS}/asio/asio/include" CACHE PATH "" FORCE)
set(ASIO_FOUND TRUE CACHE INTERNAL "")
set(CROW_ENABLE_SSL OFF CACHE BOOL "" FORCE)
set(CROW_ENABLE_COMPRESSION OFF CACHE BOOL "" FORCE)
add_subdirectory(.. core)
